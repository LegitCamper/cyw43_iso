cmake_minimum_required(VERSION 3.13)

set(PICO_BOARD pico_w)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)

project(picow_btrom_dump C CXX ASM)

    target_include_directories(cybt_shared_bus BEFORE INTERFACE ${CYW43_CUSTOM_BTFW_DIR})
endif()

    picow_btrom_dump.c
)

    pico_btstack_ble
    pico_btstack_cyw43

)

    ${CMAKE_CURRENT_LIST_DIR}
)

target_compile_definitions(picow_btrom_dump PRIVATE
    PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=10000
)

pico_enable_stdio_usb(picow_btrom_dump 1)
pico_enable_stdio_uart(picow_btrom_dump 0)

option(STATE7_PROBE "Enable bounded state-7 lifecycle experiment" OFF)
option(EVENT_PHASE_PROBE "Enable bounded scheduler callback phase experiment" OFF)
option(EVENT_PHASE_TIMING "Capture phase-2 deadline minus stored clock after stock execution" OFF)
option(EVENT_PHASE_RADIO "Capture payload mode and TX headers after stock phase 1" OFF)
option(SRAM_EXEC_PROBE "Allocate resident SRAM and check a position-independent HCI callback" OFF)
option(CONTROLLER_TIMER_PROBE "Check resident periodic timer callbacks and cancellation" OFF)
option(CONTROLLER_EVENT_PROBE "Check one resident compact scheduler event lifecycle" OFF)
option(CONTROLLER_TX_PROBE "Check custom SRAM payload through stock test-radio event" OFF)
option(CONTROLLER_RX_PROBE "Snapshot one stock DTM receive payload in resident SRAM" OFF)
option(CONTROLLER_RX_BIS_PROBE "Snapshot one fixed-channel whitened BIS-identity payload" OFF)
option(CONTROLLER_RX_ISO_PROBE "Deliver one fixed-channel RX payload through native HCI ISO" OFF)
option(CONTROLLER_BIS_PROBE "Test BIS identity and whitening through stock test-radio event" OFF)
option(CONTROLLER_AUDIO_PROBE "Test rotating LC3 payloads at nominal 10ms intervals" OFF)
option(CONTROLLER_HOP_PROBE "Test CSA2 hopping with rotating LC3 payloads" OFF)
option(CONTROLLER_DISCOVERY_PROBE "Test extended and periodic advertising before BIS audio" OFF)
option(CONTROLLER_DISCOVERY_SCAN_PROBE "Announce on all three primary advertising channels" OFF)
option(CONTROLLER_NATIVE_ISO_PROBE "Feed broadcast audio using native HCI ISO and RF credits" OFF)
option(CONTROLLER_CONTINUOUS_ISO_PROBE "Keep discovery active while streaming 1000 native ISO SDUs" OFF)
if (CONTROLLER_CONTINUOUS_ISO_PROBE)
    set(CONTROLLER_NATIVE_ISO_PROBE ON)
endif()
if (CONTROLLER_NATIVE_ISO_PROBE)
    set(CONTROLLER_DISCOVERY_SCAN_PROBE ON)
endif()
if (CONTROLLER_DISCOVERY_SCAN_PROBE)
    set(CONTROLLER_DISCOVERY_PROBE ON)
endif()
if (CONTROLLER_DISCOVERY_PROBE)
    set(CONTROLLER_HOP_PROBE ON)
endif()
if (CONTROLLER_HOP_PROBE)
    set(CONTROLLER_AUDIO_PROBE ON)
endif()
if (CONTROLLER_AUDIO_PROBE)
    set(CONTROLLER_BIS_PROBE ON)
endif()
if (CONTROLLER_BIS_PROBE)
    set(CONTROLLER_TX_PROBE ON)
endif()
if (CONTROLLER_TX_PROBE AND (CONTROLLER_TIMER_PROBE OR CONTROLLER_EVENT_PROBE))
    message(FATAL_ERROR "TX and timer/event probes require separate builds")
endif()
if (CONTROLLER_RX_ISO_PROBE)
    set(CONTROLLER_RX_BIS_PROBE ON)
endif()
if (CONTROLLER_RX_BIS_PROBE)
    set(CONTROLLER_RX_PROBE ON)
endif()
if (CONTROLLER_RX_PROBE AND (CONTROLLER_TX_PROBE OR CONTROLLER_TIMER_PROBE OR CONTROLLER_EVENT_PROBE))
    message(FATAL_ERROR "RX and other controller probes require separate builds")
endif()
if (CONTROLLER_TIMER_PROBE AND CONTROLLER_EVENT_PROBE)
    message(FATAL_ERROR "Timer and scheduler event probes require separate builds")
endif()
if (CONTROLLER_TIMER_PROBE OR CONTROLLER_EVENT_PROBE OR CONTROLLER_TX_PROBE OR CONTROLLER_RX_PROBE)
    set(SRAM_EXEC_PROBE ON)
endif()

add_executable(picow_runtime_probe picow_runtime_probe.c)
if (CONTROLLER_RX_PROBE)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(RX_CHECK_ARGS)
    if (CONTROLLER_RX_BIS_PROBE)
        list(APPEND RX_CHECK_ARGS --bis)
        target_compile_definitions(picow_runtime_probe PRIVATE CONTROLLER_RX_BIS_PROBE=1)
    endif()
    if (CONTROLLER_RX_ISO_PROBE)
        list(APPEND RX_CHECK_ARGS --iso)
        target_compile_definitions(picow_runtime_probe PRIVATE CONTROLLER_RX_ISO_PROBE=1)
    endif()
    add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/controller_rx.h
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/../analysis/check-controller-rx.py
            ${RX_CHECK_ARGS} --output-header ${CMAKE_CURRENT_BINARY_DIR}/controller_rx.h
        DEPENDS controller_rx.S iso_rx_task.S iso_rx_emit.S ../analysis/check-controller-rx.py
            ../analysis/check-iso-rx-task.py ../work/cyw43439-btrom-clean.bin
        VERBATIM)
    target_sources(picow_runtime_probe PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/controller_rx.h)
    target_include_directories(picow_runtime_probe PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
    target_compile_definitions(picow_runtime_probe PRIVATE CONTROLLER_RX_PROBE=1)
endif()
